// TRACEUR CHIRURGICAL TEMPOREL V15 - SPÉCIALISTE TIMERS & SYNCHRO REALTIME
void spyLog(String label, Object? data) {
  try {
    final timestamp = DateTime.now().toString().split(' ').last;
   
    // Extraction propre et certifiée de la stack trace réelle
    final frames = StackTrace.current.toString().split('\n');
    String realCaller = "Origine Introuvable";
    for (var frame in frames) {
      if (!frame.contains('spyLog') && frame.contains('.dart') && !frame.contains('dart:')) {
        final cleanFrame = frame.trim();
        final matchIdx = cleanFrame.indexOf('package:');
        if (matchIdx != -1) {
          realCaller = cleanFrame.substring(matchIdx).replaceAll(')', '');
        } else {
          final parts = cleanFrame.split('/');
          realCaller = parts.length > 1 ? parts.last.replaceAll(')', '') : cleanFrame;
        }
        break;
      }
    }

    print("\n⏱️ ══════════════════════════════════════════════════════════════");
    print("║ 🛰️ [LOG SYNCHRO & TIMERS] $timestamp");
    print("║ 📍 ÉVÉNEMENT : $label");
    print("║ 📂 SOURCE RÉELLE : $realCaller");
   
    Map<String, dynamic> parsed = {};
    if (data is List && data.isNotEmpty) {
      parsed = Map<String, dynamic>.from(data.first as Map);
    } else if (data is Map) {
      parsed = Map<String, dynamic>.from(data);
    }

    if (label == "INBOUND_SQL_UPDATE" || label == "RECOVERY_POLL_DATA" || label == "SQL_UPDATE_EXECUTE") {
      final v = parsed['version'] ?? '0';
      final status = parsed['game_status'] ?? 'N/A';
      final contract = parsed['contrat'] ?? 'NULL';
      final turnEnds = parsed['turn_ends_at'];
      final pIndex = parsed['phone_index'] ?? 'UNKNOWN';
     
      print("║ 📜 ÉTAT SQL -> Ver: $v | Statut: $status | Contrat: $contract | Expéditeur: $pIndex");
      if (turnEnds != null) {
        final serverTime = DateTime.parse(turnEnds as String).toUtc();
        final localDiff = serverTime.difference(DateTime.now().toUtc()).inSeconds;
        print("║ ⏳ Clé Temporaire [turn_ends_at] : $turnEnds (Expire dans: ${localDiff}s en UTC)");
      } else {
        print("║ ⚠️ Alerte Temporelle : [turn_ends_at] est ABSENT du paquet.");
      }
    } else if (parsed.isNotEmpty) {
      parsed.forEach((key, value) => print("║ 📦 $key : $value"));
    } else {
      print("║ 📦 INFORMATION : $data");
    }
    print("╚══════════════════════════════════════════════════════════════\n");
  } catch (e) {
    print("❌ Erreur interne du traceur d'espionnage : $e");
  }
}

class MultiplayerManager {
  final TarotGameLogic gameLogic;
  RealtimeChannel? _channel;
  Timer? _syncTimer;
  Timer? _recoveryTimer;
  Timer? _pingTimer;
  bool isHost = false;
  String deviceLabel = "INIT";
  String currentTable = "";
  Map<String, dynamic> currentTableData = {};
  DateTime? forcedExpiration;
  bool _isResetting = false;
  String _lastPayloadChecksum = "";

  String get _getDeviceName {
    if (kIsWeb) return "Web-Nav";
    return "${Platform.operatingSystem}-${Platform.localHostname}";
  }

  MultiplayerManager(this.gameLogic) {
    gameLogic.multiplayerManager = this;
    gameLogic.onActionBroadcast = (type, data) {
      if (_isResetting) return;

      spyLog("LOGIC_BROADCAST_TRIGGER", {
        "type": type,
        "isMultiplayer": gameLogic.isMultiplayer,
        "currentTable": currentTable,
        "currentState": gameLogic.currentState.toString()
      });

      if (!gameLogic.isMultiplayer || currentTable.isEmpty) {
        return;
      }
     
      _handleLogicBroadcast(type as String, Map<String, Object?>.from(data as Map));
      _syncActionToSql(type as String, Map<String, Object?>.from(data as Map));
    };
  }

  String get _myUid => Supabase.instance.client.auth.currentUser?.id ?? "no_id";
  String get _myName => Supabase.instance.client.auth.currentUser?.userMetadata?['full_name']
      ?? Supabase.instance.client.auth.currentUser?.userMetadata?['name']
      ?? Supabase.instance.client.auth.currentUser?.email?.split('@')[0]
      ?? "Joueur";
  String? get _myAvatar => Supabase.instance.client.auth.currentUser?.userMetadata?['avatar_url']
      ?? Supabase.instance.client.auth.currentUser?.userMetadata?['picture'];

  Future<void> joinTable(String tableId) async {
    final myId = _myUid;
    if (myId == "no_id") {
      spyLog("AUTH_ERROR", "Impossible de rejoindre : UID absent.");
      return;
    }
    try {
      spyLog("JOIN_ATTEMPT_RPC", {"tableId": tableId, "user": _myName, "uid": myId});
     
      await Future.delayed(const Duration(milliseconds: 500));
     
      final response = await Supabase.instance.client.rpc(
        'join_or_create_tarot_table',
        params: {'p_user_id': myId, 'p_user_name': _myName, 'p_user_avatar': _myAvatar}
      ).timeout(const Duration(seconds: 10));

      if (response != null) {
        final Map<String, dynamic> data;
        if (response is List && response.isNotEmpty) {
          data = Map<String, dynamic>.from(response[0]);
        } else if (response is Map) {
          data = Map<String, dynamic>.from(response);
        } else {
          spyLog("RPC_INVALID_RESPONSE_FORMAT", response);
          return;
        }
       
        currentTable = data['assigned_table_id']?.toString() ?? "";
        int mySlot = int.tryParse(data['assigned_slot']?.toString() ?? "0") ?? 0;
       
        gameLogic.tableId = currentTable;
        gameLogic.myMultiplayerSide = mySlot;
        isHost = (mySlot == 0);
        gameLogic.isHost = isHost;
        gameLogic.isMultiplayer = true;
        deviceLabel = isHost ? "ALPHA" : "BETA";
       
        spyLog("POSITION_VALIDEE", {
          "table": currentTable,
          "slot": mySlot,
          "host": isHost,
          "label": deviceLabel
        });

        gameLogic.players = List.generate(5, (i) {
          if (i == mySlot) return Player(id: i, name: _myName, isHuman: true, hand: []);
          return Player(id: i, name: "IA $i", isHuman: false, hand: []);
        });

        _connectRealtime(currentTable);
        _startPingSignal();
       
        _startHostTimer();
        if (!isHost) {
          _startBetaRecoveryPoll();
        }
        gameLogic.notifyListeners();
      }
    } catch (e) {
      spyLog("RPC_ERROR_CRITICAL", e.toString());
    }
  }

  Future<void> _forceDebugLog(String type, Map<String, Object?> updates, {String reason = "N/A", int inVer = 0, String outcome = "SUCCESS"}) async {
    try {
      String h0 = gameLogic.players.length > 0 ? gameLogic.players[0].hand.map((c) => c.id).toList().toString() : "[]";
      String h1 = gameLogic.players.length > 1 ? gameLogic.players[1].hand.map((c) => c.id).toList().toString() : "[]";
      String h2 = gameLogic.players.length > 2 ? gameLogic.players[2].hand.map((c) => c.id).toList().toString() : "[]";
      String h3 = gameLogic.players.length > 3 ? gameLogic.players[3].hand.map((c) => c.id).toList().toString() : "[]";
      String h4 = gameLogic.players.length > 4 ? gameLogic.players[4].hand.map((c) => c.id).toList().toString() : "[]";

      await Supabase.instance.client.from('debug_sync_logs').insert({
        'table_id': currentTable,
        'game_status': updates['game_status'] ?? _mapStateToStatus(gameLogic.currentState),
        'version': updates['version'] ?? gameLogic.version,
        'current_player_index': gameLogic.currentPlayerIndex,
        'phone_index': "$deviceLabel-${_getDeviceName}",
        'turn_ends_at': updates['turn_ends_at'] ?? forcedExpiration?.toIso8601String(),
        'diff_sec': "Action: $type",
        'reason_trigger': type,
        'local_state': gameLogic.currentState.toString(),
        'incoming_version': inVer,
        'local_version': gameLogic.version,
        'hand_0_state': h0,
        'hand_1_state': h1,
        'hand_2_state': h2,
        'hand_3_state': h3,
        'hand_4_state': h4,
        'validation_outcome': outcome
      });
    } catch (e) {
      spyLog("DEBUG_LOG_PUSH_ERROR", e.toString());
    }
  }

  Future<void> _syncActionToSql(String type, Map<String, Object?> data) async {
    if (!gameLogic.isMultiplayer || currentTable.isEmpty || currentTable == "salle_par_defaut" || _isResetting) return;
   
    if (gameLogic.currentState == GameState.idle && gameLogic.tricksCount > 0) {
      spyLog("SYNC_BLOCKED_IDLE", "Refus d'envoi IDLE car une partie est déjà lancée.");
      return;
    }

    if (!isHost && gameLogic.currentState == GameState.bidding && type != 'MAKE_BID') {
      try {
        final check = await Supabase.instance.client.from('table_tarot_5').select('game_status').eq('table_id', currentTable).maybeSingle();
        if (check != null && (check['game_status'] == 'STARTED' || check['game_status'] == 'DISCARDING')) {
          spyLog("CANCEL_OUTBOUND_ROLLBACK", "Blocage de l'envoi client pour préserver le STARTED de l'hôte.");
          return;
        }
      } catch (_) {}
    }
   
    gameLogic.version++;

    Map<String, Object?> updates = {
      'table_id': currentTable,
      'version': gameLogic.version,
      'last_action_time': DateTime.now().toUtc().toIso8601String(),
      'current_player_index': gameLogic.currentPlayerIndex,
      'phone_index': _getDeviceName,
      'player_count': 5,
      'game_status': _mapStateToStatus(gameLogic.currentState),
      'contrat': gameLogic.currentContractName,
      'bid_taker_id': gameLogic.takerId?.toString(),
      'partner_id': gameLogic.partnerId?.toString(),
      'called_king': gameLogic.calledKingId,
      'current_trick': gameLogic.currentTrick,
      'last_winner_index': gameLogic.lastWinnerIndex,
      'last_card_played': gameLogic.lastCardPlayed,
    };

    if (isHost) {
      forcedExpiration = DateTime.now().toUtc().add(const Duration(seconds: 30));
      updates['turn_ends_at'] = forcedExpiration!.toIso8601String();
    }

    if (gameLogic.players.length >= 5) {
      Map<String, dynamic> handsMap = {};
     
      Map<String, dynamic>? currentSnap;
      try {
        currentSnap = await Supabase.instance.client.from('table_tarot_5').select().eq('table_id', currentTable).maybeSingle();
      } catch (_) {}

      for(int i=0; i<5; i++) {
        List<int> hIds = gameLogic.players[i].hand.map((c) => c.id).toList();
        updates['hand_$i'] = hIds;
       
        String? existingUid = currentSnap?['player${i}_uid']?.toString();
        String? existingName = currentSnap?['player${i}_name']?.toString();
        if (existingUid != null && existingUid != "IA_ID" && existingName != null && i != gameLogic.myMultiplayerSide) {
          updates['player${i}_name'] = existingName;
          gameLogic.players[i].name = existingName;
        } else {
          updates['player${i}_name'] = gameLogic.players[i].name;
        }
       
        handsMap[i.toString()] = hIds;
      }
      updates['hands'] = handsMap;
      int side = gameLogic.myMultiplayerSide;
      if (side >= 0 && side < 5) {
        updates['player${side}_uid'] = _myUid;
        updates['player${side}_avatar'] = _myAvatar;
        updates['player${side}_name'] = _myName;
      }
    }

    if (type == 'MAKE_BID' || gameLogic.currentState == GameState.bidding) {
      if (gameLogic.takerId != null && gameLogic.takerId != -1) {
        updates['contrat'] = gameLogic.currentContractName;
        updates['bid_taker_id'] = gameLogic.takerId?.toString();
      } else {
        updates['contrat'] = null;
        updates['bid_taker_id'] = null;
      }
     
      if (updates['contrat'] == null && gameLogic.lastBidName.isEmpty && type != 'MAKE_BID') {
        return;
      }
    }

    final Map<String, Object?> payloadForChecksum = Map<String, Object?>.from(updates);
    payloadForChecksum.remove('last_action_time');
    payloadForChecksum.remove('turn_ends_at');

    final String currentPayloadString = jsonEncode(payloadForChecksum);
    if (currentPayloadString == _lastPayloadChecksum) {
      return;
    }
    _lastPayloadChecksum = currentPayloadString;

    try {
      spyLog("SQL_UPDATE_EXECUTE", updates);
      await Supabase.instance.client.from('table_tarot_5').update(updates).eq('table_id', currentTable);
      await _forceDebugLog("OUTBOUND_SYNC_$type", updates, outcome: "OUTBOUND_UPDATE_OK");
    } catch (e) {
      spyLog("SQL_PUSH_ERROR", e.toString());
    }
  }

  String _mapStateToStatus(GameState state) {
    switch (state) {
      case GameState.bidding: return 'BIDDING';
      case GameState.discarding: return 'DISCARDING';
      case GameState.calling: return 'CALLING';
      case GameState.playing: return 'STARTED';
      case GameState.scoring: return 'SCORING';
      default: return 'STARTED';
    }
  }

  void _startPingSignal() {
    _pingTimer?.cancel();
    _pingTimer = Timer.periodic(const Duration(seconds: 3), (timer) async {
      if (currentTable.isEmpty || !gameLogic.isMultiplayer) return;
      try {
        await Supabase.instance.client.from('table_tarot_5').update({
          'last_ping': DateTime.now().toUtc().toIso8601String(),
          'phone_index': _getDeviceName
        }).eq('table_id', currentTable);
      } catch (e) {
        // Silencieux
      }
    });
  }

 void _connectRealtime(String tableId) {
    if (tableId.isEmpty) return;
    spyLog("REALTIME_CONNECTION_START", {"tableId": tableId});
    _channel = Supabase.instance.client.channel(tableId);
   
    _channel!.onPostgresChanges(
      event: PostgresChangeEvent.update,
      schema: 'public',
      table: 'table_tarot_5',
      callback: (payload) {
        final Map<String, Object?> newData = Map<String, Object?>.from(payload.newRecord);
       
        if (newData['phone_index'] == _getDeviceName) {
          spyLog("INBOUND_IGNORED_SELF_SOURCE", "Le paquet provient de cet appareil même. Traitement ignoré.");
          return;
        }

        final String? statusEnBase = newData['game_status']?.toString();
        if (statusEnBase == 'IDLE' || statusEnBase == null || statusEnBase.isEmpty) {
          spyLog("INBOUND_REJECTED_STATUS_IDLE_OR_EMPTY", {"status": statusEnBase});
          return;
        }

        final int versionSql = int.tryParse(newData['version']?.toString() ?? '0') ?? 0;
       
        if (versionSql <= gameLogic.version) {
          spyLog("INBOUND_VERSION_LOW_WARNING", "Version SQL ($versionSql) <= Version Locale (${gameLogic.version}). Vérification forcée du changement de statut.");
          if (!isHost && statusEnBase != _mapStateToStatus(gameLogic.currentState)) {
            spyLog("FORCED_STATUS_ALIGNMENT_DESPITE_VERSION", "Statut en base différent du statut local actuel (${gameLogic.currentState}). Alignement de force.");
          } else {
            _forceDebugLog("INBOUND_REJECTED", newData, inVer: versionSql, outcome: "REJECTED_VERSION_TOO_LOW_NO_STATUS_CHANGE");
            return;
          }
        }
       
        gameLogic.version = versionSql;

        spyLog("INBOUND_SQL_UPDATE", newData);

        print("🕵️ [MOUCHARD 1 - CONFLIT HORLOGE] Réception brute SQL.");
        print(" -> Appareil Émetteur (phone_index) : ${newData['phone_index']}");
        print(" -> Horodatage de fin distant (turn_ends_at) : ${newData['turn_ends_at']}");
        print(" -> Heure locale de cet appareil : ${DateTime.now().toUtc().toIso8601String()}");
        if (newData['turn_ends_at'] != null) {
          final DateTime distEnd = DateTime.parse(newData['turn_ends_at'] as String).toUtc();
          final int diffCalculated = distEnd.difference(DateTime.now().toUtc()).inSeconds;
          print(" -> Résultat du calcul de soustraction locale : ${diffCalculated}s restantes.");
          if (diffCalculated > 28) {
            print(" -> 🚨 CONSTATION MOUCHARD 1 : Le téléphone s'est réaligné sur 30s complètes car l'émetteur a écrasé l'heure avec NOW() + 30s au lieu de propager la décrémentation en cours !");
          }
        }

        double targetRemainingTime = gameLogic.remainingTime;
        if (newData['turn_ends_at'] != null) {
          forcedExpiration = DateTime.parse(newData['turn_ends_at'] as String).toUtc();
          final int localDiff = forcedExpiration!.difference(DateTime.now().toUtc()).inSeconds;
          if ((gameLogic.remainingTime - localDiff).abs() > 2.0) {
            targetRemainingTime = localDiff.clamp(0, 30).toDouble();
            gameLogic.remainingTime = targetRemainingTime;
          }
        }

        print("🕵️ [MOUCHARD 2 - EXCLUSION TABLE ET CONTRAT]");
        print(" -> Mon ID de slot local assigné : ${gameLogic.myMultiplayerSide}");
        print(" -> Contrat lu en base : ${newData['contrat']} | Preneur en base : ${newData['bid_taker_id']}");
        print(" -> Roi appelé lu en base (called_king) : ${newData['called_king']}");
        print(" -> Partenaire lu en base (partner_id) : ${newData['partner_id']}");
        for(int idx = 0; idx < 5; idx++) {
          print(" -> Position SQL slot $idx : uid=${newData['player${idx}_uid']} | name=${newData['player${idx}_name']}");
        }
        if (newData['player${gameLogic.myMultiplayerSide}_uid'] != _myUid) {
          print(" -> 🚨 CONSTATATION MOUCHARD 2 : Mon UID local n'est pas associé à mon slot dans cette ligne SQL entrante ! Écrasement ou désynchronisation détectée !");
        }

        if (!isHost) {
          if (newData['game_status'] == 'DISCARDING') {
            gameLogic.currentState = GameState.discarding;
          } else if (newData['game_status'] == 'CALLING') {
            gameLogic.currentState = GameState.calling;
          } else if (newData['game_status'] == 'STARTED') {
            if (gameLogic.players.isNotEmpty && gameLogic.players[gameLogic.myMultiplayerSide].hand.isNotEmpty) {
              gameLogic.currentState = GameState.playing;
            }
          } else if (newData['game_status'] == 'BIDDING') {
            gameLogic.currentState = GameState.bidding;
          } else if (newData['game_status'] == 'SCORING') {
            gameLogic.currentState = GameState.scoring;
          }
          spyLog("BETA_NEW_STATE_APPLIED", gameLogic.currentState.toString());
        }

        if (!isHost && newData['game_status'] == 'DISCARDING' && gameLogic.currentState == GameState.bidding) {
          spyLog("BETA_DISCARD_SWITCH", "Bascule propre : Détection de la phase d'écart DISCARDING.");
          _forceDebugLog("INBOUND_ACCEPT_DISCARD", newData, inVer: versionSql, outcome: "SWITCH_TO_DISCARD");
          gameLogic.handleAction('SYNC_VIA_SQL', newData);
          gameLogic.remainingTime = targetRemainingTime;
          gameLogic.refresh();
          return;
        }

        if (!isHost && newData['game_status'] == 'STARTED' && gameLogic.currentState != GameState.playing) {
          if (gameLogic.players.isNotEmpty && gameLogic.players[gameLogic.myMultiplayerSide].hand.isNotEmpty) {
            spyLog("BETA_AUTO_SWITCH", "Bascule forcée : Détection du statut STARTED dans le flux SQL.");
            _forceDebugLog("INBOUND_ACCEPT_STARTED", newData, inVer: versionSql, outcome: "SWITCH_TO_STARTED");
            gameLogic.handleAction('GAME_STARTED_TRIGGER', newData);
            gameLogic.remainingTime = targetRemainingTime;
          }
        }

        if (isHost) {
          if (_isResetting || (newData['contrat'] != null && gameLogic.currentContractName == null && gameLogic.takerId == null)) {
            spyLog("HOST_REJECT_RESURRECTION", "Refus de synchronisation : Contrat zombie détecté.");
            _forceDebugLog("HOST_ZOMBIE_BLOCKED", newData, inVer: versionSql, outcome: "HOST_ZOMBIE_BLOCKED");
            return;
          }
          if (newData['contrat'] != null && newData['contrat'] != gameLogic.lastBidName) {
            spyLog("HOST_ALIGNS_CONTRAT", {"from": gameLogic.lastBidName, "to": newData['contrat']});
            gameLogic.lastBidName = (newData['contrat'] as String?) ?? "";
          }
         
          gameLogic.handleAction('SYNC_VIA_SQL', newData);
          gameLogic.refresh();
         
          if (newData['game_status'] == 'STARTED') {
            spyLog("HOST_FORCES_BROADCAST_START", "L'Hôte force l'envoi de la bascule vers STARTED.");
            _handleLogicBroadcast('GAME_STARTED_TRIGGER', {'version': versionSql});
          }
          return;
        }

        if (newData['pli_joueurs'] is List) {
          newData['pli_joueurs'] = {'0': [], '1': [], '2': [], '3': [], '4': []};
        }
       
        if (gameLogic.players.length >= 5) {
          for(int i=0; i<5; i++) {
            String nameKey = 'player${i}_name';
            if (newData[nameKey] != null && gameLogic.myMultiplayerSide != i) {
              gameLogic.players[i].name = newData[nameKey] as String;
            }
          }
        }

        if (newData['sync_lock'] == true) {
          spyLog("SYNC_IGNORED", "sync_lock est actif.");
          _forceDebugLog("INBOUND_SYNC_LOCKED", newData, inVer: versionSql, outcome: "SYNC_LOCK_TRUE_IGNORED");
          return;
        }

        if (newData['contrat'] == '') {
          newData['contrat'] = null;
        }

        _forceDebugLog("INBOUND_PROCESS_END", newData, inVer: versionSql, outcome: "PROCESSED_AND_NOTIFIED");
        gameLogic.handleAction('SYNC_VIA_SQL', newData);
        gameLogic.remainingTime = targetRemainingTime;
        gameLogic.refresh();
      },
    ).onBroadcast(event: 'game_action', callback: (payload) {
      if (payload == null || payload['senderId'] == _myUid) return;
      spyLog("BROADCAST_RECEIVED", {"type": payload['type'], "from": payload['senderId']});
     
      final data = Map<String, Object?>.from(payload['data'] as Map? ?? {});
      if (payload['type'] == 'GAME_STARTED_TRIGGER') {
        spyLog("BETA_RECEIVES_FORCED_START", "Téléphone invité forcé de basculer l'affichage des cartes.");
      }
     
      gameLogic.handleAction(payload['type'] as String? ?? 'UNKNOWN', data);
      gameLogic.refresh();
    }).onPresenceSync((p) {
      spyLog("PRESENCE_SYNC_TRIGGERED", "Mise à jour des slots de présence.");
      _syncAvatars();
    }).subscribe((status, _) async {
      if (status == RealtimeSubscribeStatus.subscribed) {
        spyLog("REALTIME_SUBSCRIBED", "Canal $tableId connecté au réseau.");
        await _channel?.track({'id': _myUid, 'name': _myName, 'slot': gameLogic.myMultiplayerSide, 'avatar': _myAvatar});
      }
    });
  }

  void _syncAvatars() {
    if (gameLogic.players.length < 5) return;
    final states = _channel?.presenceState() ?? [];
    for (var state in states) {
      for (var m in state.presences) {
        final Map<String, Object?> payload = Map<String, Object?>.from(m.payload);
        int? pSlot = payload['slot'] as int?;
        if (payload['id'] != _myUid && pSlot != null && pSlot >= 0 && pSlot < 5) {
          gameLogic.players[pSlot].name = payload['name'] as String? ?? "Joueur";
          gameLogic.players[pSlot].isHuman = true;
          spyLog("AVATAR_FOUND", {"name": gameLogic.players[pSlot].name, "slot": pSlot});
        }
      }
    }
    gameLogic.notifyListeners();
  }

  void _handleLogicBroadcast(String type, Map<String, Object?> data) {
    if (currentTable.isEmpty || !gameLogic.isMultiplayer) return;
    _channel?.sendBroadcastMessage(event: 'game_action', payload: {'type': type, 'senderId': _myUid, 'data': data});
  }

  void _startHostTimer() {
    _syncTimer?.cancel();
    if (forcedExpiration == null) {
      forcedExpiration = DateTime.now().toUtc().add(const Duration(seconds: 30));
    }

import 'models.dart';

class ScoringLogic {
  static double getRequiredPoints(int bouts) {
    if (bouts == 3) return 36.0;
    if (bouts == 2) return 41.0;
    if (bouts == 1) return 51.0;
    return 56.0;
  }

  static Map<String, double> getLiveBalance(List<Player> players, int takerId, int? partnerId) {
    double takerTotal = 0.0;
    double defenseTotal = 0.0;
    for (var p in players) {
      double pts = 0.0;
      for (var card in p.tricksWon) { pts += card.points; }
      if (p.id == takerId || (partnerId != null && p.id == partnerId)) {
        takerTotal += pts;
      } else {
        defenseTotal += pts;
      }
    }
    return {'taker': takerTotal, 'defense': defenseTotal};
  }

  static Map<int, double> computeFinalGains({
    required List<Player> players,
    required int takerId,
    int? partnerId,
    required BidType? selectedBid,
    int? petitAuBoutWinner,
    Map<int, String>? poignees,
  }) {
    if (selectedBid == null) return {};
    var balance = getLiveBalance(players, takerId, partnerId);
    double scoreAttaque = balance['taker']!;
    int nbBouts = 0;
    for (var p in players) {
      if (p.id == takerId || (partnerId != null && p.id == partnerId)) {
        nbBouts += p.tricksWon.where((c) => c.isBout).length;
      }
    }
    double required = getRequiredPoints(nbBouts);
    double diff = scoreAttaque - required;
    double base = (25 + diff.abs());
    if (petitAuBoutWinner != null) {
      bool sideTaker = (petitAuBoutWinner == takerId || (partnerId != null && petitAuBoutWinner == partnerId));
      base += sideTaker ? 10 : -10;
    }
    if (diff < 0) base = -base;
    int mult = 1;
    switch (selectedBid) {
      case BidType.petite: mult = 1; break;
      case BidType.garde: mult = 2; break;
      case BidType.gardeSans: mult = 4; break;
      case BidType.gardeContre: mult = 6; break;
      default: mult = 1;
    }
    double finalBase = base * mult;
    double totalPrimesPoignees = 0;
    if (poignees != null) {
      poignees.forEach((pid, type) {
        double valeur = (type == "TRIPLE") ? 50 : (type == "DOUBLE") ? 30 : 20;
        bool isAnnonceurAttaque = (pid == takerId || (partnerId != null && pid == partnerId));
        if (diff >= 0) {
          totalPrimesPoignees += isAnnonceurAttaque ? valeur : -valeur;
        } else {
          totalPrimesPoignees += isAnnonceurAttaque ? -valeur : valeur;
        }
      });
    }
    double scoreParPart = finalBase + totalPrimesPoignees;
    Map<int, double> results = {};
    int nbPlayers = players.length;
    for (var p in players) {
      if (p.id == takerId) {
        if (nbPlayers == 5 && partnerId != null && partnerId != takerId) {
          results[p.id] = scoreParPart * 2;
        } else {
          results[p.id] = scoreParPart * (nbPlayers - 1);
        }
      } else if (nbPlayers == 5 && partnerId != null && p.id == partnerId) {
        results[p.id] = scoreParPart;
      } else {
        results[p.id] = -scoreParPart;
      }
    }
    return results;
  }
}
