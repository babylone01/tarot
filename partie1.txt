import 'package:flutter/material.dart'; // Correction : minuscule
import 'dart:math' as math;

class GothicDownloadAnimation extends StatefulWidget {
  const GothicDownloadAnimation({super.key});
  @override
  State<GothicDownloadAnimation> createState() => _GothicDownloadAnimationState();
}

class _GothicDownloadAnimationState extends State<GothicDownloadAnimation> with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  int _dataPiecesCut = 0;
  bool _isCutting = false;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      duration: const Duration(seconds: 3),
      vsync: this,
    )..repeat();
    
    _controller.addListener(() {
      if (_controller.value >= 0.5 && !_isCutting) {
        if (mounted) {
          setState(() {
            _isCutting = true;
            _dataPiecesCut++;
          });
        }
        
        Future.delayed(const Duration(milliseconds: 600), () {
          if (mounted) {
            setState(() => _isCutting = false);
          }
        });
      }
    });
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFF1A1A1A),
      body: Center(
        child: AnimatedBuilder(
          animation: _controller,
          builder: (context, child) {
            return Column(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                CustomPaint(
                  painter: GothicPainter(
                    progress: _controller.value,
                    isCutting: _isCutting,
                  ),
                  size: const Size(200, 200),
                ),
                const SizedBox(height: 30),
                Text(
                  "RÉCOLTE DE L'ÂME... $_dataPiecesCut ÂMES",
                  style: const TextStyle(
                    color: Color(0xFFC0C0C0),
                    fontSize: 18,
                    fontWeight: FontWeight.bold,
                    fontFamily: 'serif',
                  ),
                ),
              ],
            );
          },
        ),
      ),
    );
  }
}

class GothicPainter extends CustomPainter {
  final double progress;
  final bool isCutting;
  GothicPainter({required this.progress, required this.isCutting});

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final Paint ironPaint = Paint()
      ..color = const Color(0xFF505050)
      ..style = PaintingStyle.stroke
      ..strokeWidth = 3.0;
    final Paint goldPaint = Paint()
      ..color = const Color(0xFFD4AF37)
      ..style = PaintingStyle.fill;

    // Dessin de la Faucheuse (Capuche et Crâne)
    canvas.drawCircle(Offset(center.dx, center.dy - 55), 10, Paint()..color = Colors.white);
   
    // Animation de la Faucille
    double angle;
    if (progress < 0.5) {
      angle = -math.pi / 2 - (progress * 2 * math.pi / 3);
    } else {
      angle = -math.pi / 2 - (math.pi / 3) + ((progress - 0.5) * 2 * math.pi);
    }

    final Offset sickleBase = Offset(center.dx - 15, center.dy - 20);
    final Offset sickleEnd = Offset(
      sickleBase.dx + math.cos(angle) * 70,
      sickleBase.dy + math.sin(angle) * 70,
    );
    canvas.drawLine(sickleBase, sickleEnd, ironPaint);

    // Dessin du Jeton (L'âme)
    final Offset tokenPos = Offset(center.dx + 50, center.dy + 30);
    if (!isCutting) {
      canvas.drawCircle(tokenPos, 25, goldPaint);
    } else {
      final double sep = (progress > 0.6) ? (progress - 0.6) * 50 : 0;
      canvas.drawArc(Rect.fromCircle(center: tokenPos.translate(-sep, 0), radius: 25), math.pi / 2, math.pi, true, goldPaint);
      canvas.drawArc(Rect.fromCircle(center: tokenPos.translate(sep, 0), radius: 25), -math.pi / 2, math.pi, true, goldPaint);
    }
  }

  @override
  bool shouldRepaint(covariant GothicPainter oldDelegate) => true;
}

import 'package:tarot_app/models.dart';

class TarotAI {
  // --- MÉMOIRE VIVE ---
  static Map<int, Set<CardSuit>> chicanesGlobales = {};
  static List<CardModel> cartesTombees = [];

  static void resetMemory(int playerCount) {
    chicanesGlobales = {for (int i = 0; i < playerCount; i++) i: {}};
    cartesTombees = [];
  }

  static void analyzeTrick(List<Map<int, CardModel>> trick, CardSuit leadSuit) {
    for (var entry in trick) {
      int pid = entry.keys.first;
      CardModel card = entry.values.first;
      cartesTombees.add(card);
     
      // Détection des chicanes
      if (card.suit != leadSuit && card.suit != CardSuit.atout && card.suit != CardSuit.excuse) {
        chicanesGlobales[pid]?.add(leadSuit);
      }
    }
  }

  // --- LOGIQUE D'ENCHÈRE (SÉCURISÉE) ---
  static BidType decideBid(Player p) {
    int atouts = p.hand.where((c) => c.suit == CardSuit.atout).length;
    double points = p.hand.fold(0.0, (prev, c) => prev + c.points);
    int bouts = p.hand.where((c) => c.isBout).length;

    if ((atouts >= 10 && bouts >= 1) || (atouts >= 8 && bouts >= 2) || (atouts >= 7 && bouts == 3)) {
      return BidType.garde;
    }
   
    if (bouts >= 2 && (atouts >= 6 || points > 40)) return BidType.petite;
    if (bouts == 1 && (atouts >= 8 || (atouts >= 6 && points >= 45))) return BidType.petite;
    if (bouts == 0 && atouts >= 11) return BidType.petite;
   
    return BidType.pass;
  }

  // --- LE CERVEAU : DÉCISION DE LA CARTE ---
  static CardModel decideCardToPlay({
    required List<CardModel> playableCards,
    CardSuit? leadSuit,
    CardSuit? calledKing,
    required int myId,
    int? tId,
    int? pId,
    required List<Map<int, CardModel>> currentTrick,
    required int totalTricksPlayed,
    required int totalPlayers,
  }) {
    if (playableCards.length == 1) return playableCards.first;

    final bool isTakerSide = (myId == tId || (pId != null && myId == pId));
    final bool isLastTrick = totalTricksPlayed >= (totalPlayers == 5 ? 14 : 17);
    final bool isEndGame = totalTricksPlayed > 13;
   
    // 1. ANALYSE DE LA TABLE
    CardModel? bestOnTable;
    int? currentWinnerId;
    if (currentTrick.isNotEmpty) {
      bestOnTable = _getBestOnTable(currentTrick, leadSuit);
      currentWinnerId = currentTrick.firstWhere((e) => e.values.first == bestOnTable).keys.first;
    }

    bool allyIsMaster = false;
    if (currentWinnerId != null) {
      bool winnerIsTakerSide = (currentWinnerId == tId || (pId != null && currentWinnerId == pId));
      allyIsMaster = (isTakerSide == winnerIsTakerSide);
    }

    // --- STRATÉGIE EXCUSE (SAUVETAGE) ---
    CardModel? excuse = playableCards.where((c) => c.suit == CardSuit.excuse).firstOrNull;
    if (excuse != null && currentTrick.isNotEmpty) {
      // Si je suis preneur et que je vais perdre un gros point sur ce pli
      if (isTakerSide && !allyIsMaster && bestOnTable != null && bestOnTable.points >= 3.5) {
        return excuse;
      }
      // Si la défense joue une couleur où je suis coupé mais que je veux garder mes atouts
      if (isTakerSide && leadSuit != null && !playableCards.any((c) => c.suit == leadSuit) && !isEndGame) {
        return excuse;
      }
    }

    // 2. 📡 RADAR DE DANGER (CHICANES)
    bool willBeCutByDefense = false;
    if (leadSuit != null && isTakerSide) {
      for (int i = 0; i < totalPlayers; i++) {
        if (currentTrick.any((e) => e.containsKey(i)) || i == myId) continue;
        bool isEnemy = (i != tId && (pId == null || i != pId));
        if (isEnemy && chicanesGlobales[i]?.contains(leadSuit) == true) {
          willBeCutByDefense = true;
          break;
        }
      }
    }

    // 3. 🛡️ PROTECTION DU PETIT (STRICTE)
    CardModel? petit = playableCards.where((c) => c.suit == CardSuit.atout && c.value == "1").firstOrNull;
    if (petit != null && isTakerSide) {
      bool forced = playableCards.length == 1;
      if (!forced && !isLastTrick) {
        // On cherche n'importe quelle autre carte pour ne pas donner le Petit
        var backup = playableCards.where((c) => c != petit).toList();
        if (backup.isNotEmpty) {
          // Si on a l'Excuse, on l'utilise pour sauver le petit
          if (excuse != null) return excuse;
          backup.sort((a, b) => a.points.compareTo(b.points));
          return backup.first;
        }
      }
    }

    // 4. LOGIQUE D'ENTAME (Taker)
    if (currentTrick.isEmpty) {
      if (petit != null && excuse != null) return excuse;
     
      // CHASSE À L'ATOUT : Le preneur purge la défense
      if (isTakerSide && totalTricksPlayed < 6) {
        var trumps = playableCards.where((c) => c.suit == CardSuit.atout && c != petit).toList();
        if (trumps.length >= 4) {
          trumps.sort((a, b) => b.power.compareTo(a.power));
          return trumps.first;
        }
      }

      // ÉVITER DE JETER DES TÊTES SI DÉFENSE A ENCORE DE L'ATOUT
      var colorCards = playableCards.where((c) => c.suit != CardSuit.atout && c.suit != CardSuit.excuse).toList();
      if (colorCards.isNotEmpty) {
        colorCards.sort((a, b) => b.power.compareTo(a.power));
        
        if (isTakerSide && isEndGame) {
          // En fin de partie, on joue d'abord les petites cartes si risque de coupe
          var smallCards = colorCards.where((c) => c.points < 1.5).toList();
          if (smallCards.isNotEmpty) return smallCards.first;
        }

        // Radar chicane pour ne pas donner un Roi
        var safeChoices = colorCards.where((card) {
          for (int i = 0; i < totalPlayers; i++) {
            if (i == myId) continue;
            bool isEnemy = isTakerSide ? (i != tId && (pId == null || i != pId)) : (i == tId || (pId != null && i == pId));
            if (isEnemy && chicanesGlobales[i]?.contains(card.suit) == true) return false;
          }
          return true;
        }).toList();

        if (safeChoices.isNotEmpty) return safeChoices.first;
        return colorCards.last; // On joue la plus petite si tout est risqué
      }
    }

    // 5. TENTATIVE DE PRENDRE LA MAIN
    if (bestOnTable != null) {
      var winningCards = playableCards.where((c) => _canWin(c, bestOnTable!, leadSuit)).toList();
      if (winningCards.isNotEmpty) {
        winningCards.sort((a, b) => a.power.compareTo(b.power));
        
        // Si je suis preneur, je ne gagne avec le Petit que si c'est sûr
        if (winningCards.first == petit && !isLastTrick && winningCards.length > 1) {
          return winningCards[1];
        }
        
        // Ne pas jeter un gros atout sur un pli "pauvre"
        if (isTakerSide && winningCards.first.suit == CardSuit.atout && bestOnTable.points < 1.0 && !isEndGame) {
           var throwAway = playableCards.where((c) => c.suit != CardSuit.atout).toList();
           if (throwAway.isNotEmpty) {
             throwAway.sort((a, b) => a.points.compareTo(b.points));
             return throwAway.first;
           }
        }

        return winningCards.first;
      }
    }

    // 6. ALLIÉ MAÎTRE
    if (allyIsMaster) {
      if (isLastTrick && petit != null) return petit;
      // On charge en points si pas de risque de coupe
      if (!willBeCutByDefense) {
        playableCards.sort((a, b) => b.points.compareTo(a.points));
        return playableCards.first;
      }
    }

    // 7. DÉFAUSSE / JEU PAR DÉFAUT
    List<CardModel> choices = List.from(playableCards);
    if (choices.length > 1 && petit != null) choices.remove(petit);
    
    choices.sort((a, b) => a.points.compareTo(b.points));
    if (choices.isNotEmpty && choices.first.suit == CardSuit.atout && choices.length > 1) {
      choices.sort((a, b) => a.power.compareTo(b.power));
    }

    return choices.isEmpty ? playableCards.first : choices.first;
  }

  static CardModel? _getBestOnTable(List<Map<int, CardModel>> trick, CardSuit? lead) {
    if (trick.isEmpty) return null;
    CardModel best = trick.first.values.first;
    for (var i = 1; i < trick.length; i++) {
      CardModel current = trick[i].values.first;
      if (_canWin(current, best, lead)) best = current;
    }
    return best;
  }

  static bool _canWin(CardModel challenger, CardModel currentBest, CardSuit? lead) {
    if (challenger.suit == CardSuit.excuse) return false;
    if (currentBest.suit == CardSuit.atout) {
      if (challenger.suit != CardSuit.atout) return false;
      return challenger.power > currentBest.power;
    }
    if (challenger.suit == CardSuit.atout) return true;
    if (challenger.suit != lead) return false;
    return challenger.power > currentBest.power;
  }

  static void feedbackFinDeManche(bool gain, double score) {}

  static bool shouldCallKing(List<CardModel> hand) {
    return hand.where((c) => c.suit == CardSuit.atout).length > 6;
  }
}

  void _sortHand(List<CardModel> h) {
    h.sort((a, b) {
      int p(CardSuit s) {
        switch (s) {
          case CardSuit.atout: case CardSuit.excuse: return 0;
          case CardSuit.coeur: return 1; case CardSuit.pique: return 2;
          case CardSuit.carreau: return 3; case CardSuit.trefle: return 4;
        }
      }
      return p(a.suit) != p(b.suit) ? p(a.suit).compareTo(p(b.suit)) : b.power.compareTo(a.power);
    });
    for (var c in h) {
      c.assetPath = _getAssetPath(c.suit, c.value);
    }
  }

  String _getAssetPath(CardSuit s, String v) {
  // 1. On gère l'Excuse d'abord
  if (s == CardSuit.excuse) {
    return "assets/images/cards/L'Excuse.png";
  }

  // 2. On prépare le nom de l'enseigne (sn)
  String sn = s.toString().split('.').last.toLowerCase();
  sn = sn.replaceAll('trefle', 'Trefle')
         .replaceAll('carreau', 'Carreau')
         .replaceAll('coeur', 'Coeur')
         .replaceAll('pique', 'Pique')
         .replaceAll('atout', 'Atout');

  // 3. On prépare le nom de la valeur (vn)
  String vn = v.replaceAll('R', 'Roi')
               .replaceAll('D', 'Dame')
               .replaceAll('C', 'Cavalier')
               .replaceAll('V', 'Valet');

  // 4. On renvoie le chemin final
  return "assets/images/cards/${vn.isEmpty ? '?' : vn} de ${sn.isEmpty ? '?' : sn}.png";
}

   void _iaSmartDiscard() {
    if (takerId == null) return;
    List<CardModel> h = players[takerId!].hand;
    currentEcartVisual.clear();
    int n = (players.length == 5 ? 3 : 6);
   
    // 1. On cherche les coupes (couleurs où on a peu de cartes, hors Rois)
    List<CardSuit> suits = [CardSuit.pique, CardSuit.coeur, CardSuit.carreau, CardSuit.trefle];
    suits.sort((a, b) => h.where((c) => c.suit == a).length.compareTo(h.where((c) => c.suit == b).length));

    for (var s in suits) {
      var candidates = h.where((c) => c.suit == s && c.value != "R" && !c.isBout).toList();
      for (var c in candidates) {
        if (currentEcartVisual.length < n) {
          currentEcartVisual.add(c);
        }
      }
    }

    // 2. Si pas assez de cartes, on prend les plus petites cartes restantes (hors Atout/Roi)
    if (currentEcartVisual.length < n) {
      List<CardModel> remain = h.where((c) => !currentEcartVisual.contains(c) && c.suit != CardSuit.atout && c.value != "R").toList();
      remain.sort((a, b) => a.points.compareTo(b.points));
      while (currentEcartVisual.length < n && remain.isNotEmpty) {
        currentEcartVisual.add(remain.removeAt(0));
      }
    }

    // 3. Suppression physique et propre des cartes de la main
    for (var c in currentEcartVisual) {
      h.removeWhere((card) => card.id == c.id);
    }

    _sortHand(players[takerId!].hand);
    _finishDiscard();
  }

  void _activateRealtimeSync() {
    _realtimeSubscription?.cancel();
   
    debugPrint("🕵️ ESPION : Démarrage du flux sur table: $tableId");

    _realtimeSubscription = _supabase
        .from('table_tarot_5')
        .stream(primaryKey: ['table_id'])
        .eq('table_id', tableId!)
        .listen((data) {
     
      if (data.isEmpty) return;

      final latest = data.last;
     
      // PROTECTION ANTI-SPAM : Si la version reçue n'est pas supérieure à notre version locale,
      // on ignore le paquet pour éviter de traiter en boucle la même information.
      final int ver = latest['version'] ?? 0;
      if (ver <= version && currentState != GameState.idle) {
        return;
      }
     
      final String stat = latest['game_status'] ?? 'IDLE';

      if (ver == 0 || stat == 'IDLE') {
        if (currentState != GameState.scoring) {
          debugPrint("🧹 ESPION : Reset de la manche détecté");
          _stopTimer();
          players.clear();
          currentTrickData.clear();
          lastBidName = "";
          selectedBid = null;
          calledKingSuit = null;
          currentState = GameState.idle;
          notifyListeners();
        }
        return;
      }

      // --- RECTIFICATION : LECTURE DE LA COLONNE DE TEMPS DEPUIS TURN_ENDS_AT ---
      if (latest['turn_ends_at'] != null) {
        DateTime expiration = DateTime.parse(latest["turn_ends_at"].toString()).toUtc();
        Duration reste = expiration.difference(DateTime.now());
        double secondesServeur = math.min(30.0, reste.inMilliseconds / 1000.0);

        // On ne met à jour le chrono que si l'écart est flagrant (plus de 1.5 seconde de décalage)
        if ((secondesServeur - _remainingTime).abs() > 1.5) {
          _remainingTime = secondesServeur > 0 ? secondesServeur : 0.0;
          _timeController.add(_remainingTime);
        }
      }

      // Mise à jour de la version locale avant d'appliquer les données
      version = ver;

      syncCommonData(latest);
     
      if (isMultiplayer && latest['last_card_played'] != null && latest['last_card_played'] != "") {
        _handleRemoteSqlAction(latest);
      }
    }, onError: (error) {
      debugPrint("❌ CRASH DU FLUX : $error");
    });
  }






  void _handleRemoteSqlAction(Map<String, dynamic> data) {

    // car mon interface l'a déjà exécutée localement.

      String cardStr = data['last_card_played'].toString();
      if (cardStr.isEmpty) return;

        String val = parts[0];

  void handleAction(String type, Map<String, dynamic> data) async {
    try {
      if (type == 'SYNC_VIA_SQL') {
        final String? effectiveId = data['table_id']?.toString() ?? tableId;
        if (isMultiplayer && effectiveId != null) {
          final res = await _supabase.from('table_tarot_5').select().eq('table_id', effectiveId).order('created_at', ascending: false).limit(1).single();
          syncCommonData(res);
         
          if (res['hands'] != null) {
            Map<String, dynamic> handsMap = {};
            var rawHands = res['hands'];
           
            if (rawHands is String) {
              // Nettoyage manuel de la String pour virer les antislashs et guillemets en trop
              String cleanStr = rawHands.replaceAll('\\"', '"');
              if (cleanStr.startsWith('"') && cleanStr.endsWith('"')) {
                cleanStr = cleanStr.substring(1, cleanStr.length - 1);
              }
             
              // Extraction de ta liste de cartes sans import bloquant
              String recherche = '"$myMultiplayerSide":[';
              if (cleanStr.contains(recherche)) {
                int debut = cleanStr.indexOf(recherche) + recherche.length;
                int fin = cleanStr.indexOf(']', debut);
                String listeCartes = cleanStr.substring(debut, fin);
                if (listeCartes.isNotEmpty) {
                  List<int> cardIds = listeCartes.split(',').map((e) => int.parse(e.trim())).toList();
                  handsMap[myMultiplayerSide.toString()] = cardIds;
                }
              }
            } else if (rawHands is Map) {
              handsMap = Map<String, dynamic>.from(rawHands);
            }

            syncReceivedCards({'to': myMultiplayerSide, 'h': handsMap[myMultiplayerSide.toString()]});
          }
        }
      }
      else if (type == 'MAKE_BID') {
        int pid = int.tryParse(data['pid']?.toString() ?? '0') ?? 0;
        receiveBid(data['bid']?.toString() ?? "pass", pid);
      }
      else if (type == 'SELECT_KING') {
        int suitIdx = int.tryParse(data['suit']?.toString() ?? '0') ?? 0;
        selectCalledKing(CardSuit.values[suitIdx]);
      }
      else if (type == 'PLAY_CARD') {
        int pid = int.tryParse(data['pid']?.toString() ?? '0') ?? 0;
        final all = generateFullTarotDeck();
        final card = all.firstWhere((c) => c.suit.toString() == data['cardSuit'] && c.value == data['cardValue'].toString());
        playCard(pid, card);
      }
    } catch (e) {
      debugPrint("❌ Erreur handleAction: $e");
    }
  }





    void syncCommonData(Map<String, dynamic> data) {
    // --- VERROU ANTI-DISPARITION ET CLIGNOTEMENT ---
    // Si on est en train de regarder le chien ou de faire l'écart, on bloque le réseau
    // pour éviter qu'il n'écrase notre affichage local et ne fasse sauter les cartes.
    if (currentState == GameState.dogRevealed || currentState == GameState.discarding) {
      return;
    }

    // --- 🕵️ ANALYSE DU CYCLE (WEEKEND -> BIDDING -> START-UP) ---
    final String sqlStatus = data['game_status']?.toString() ?? 'IDLE';
    final String sqlContrat = data['contrat']?.toString() ?? '';
   
    // 1. GESTION DU CYCLE LOGIQUE
    if (sqlStatus == 'WAITING') {
      // MODE WEEKEND (Attente)
      currentState = GameState.idle;
    }
    // --- On ne repasse en BIDDING QUE si on n'est pas déjà en jeu ---
    else if (sqlStatus == 'BIDDING' && currentState != GameState.playing && currentState != GameState.discarding && currentState != GameState.calling) {
      currentState = GameState.bidding;
      if (sqlContrat == '' || sqlContrat == 'null') {
        lastBidName = "";
        selectedBid = null;
      }
    }
    // --- INTERCEPTION PROPRE DE LA PHASE D'ÉCART POUR LE BETA ---
    else if (sqlStatus == 'DISCARDING') {
      if (currentState != GameState.scoring) {
        currentState = GameState.discarding;
      }
    }

    else if (sqlStatus == 'CALLING') {
      if (currentState != GameState.scoring) {
        currentState = GameState.calling;
      }
    }

    else if (sqlStatus == 'STARTED' || (sqlContrat != '' && sqlContrat != 'null' && sqlStatus != 'DISCARDING')) {
      // FORCE LE PASSAGE AU JEU : Si statut STARTED OU si un contrat est présent (et hors phase écart)
      if (currentState != GameState.scoring) {
        currentState = GameState.playing;
      }
    }

    // --- On gère UNIQUEMENT le score ---
    else if (sqlStatus == 'SCORING') {
      currentState = GameState.scoring;
    }

    // 2. SYNCHRONISATION TECHNIQUE
    int pc = int.tryParse(data['player_count']?.toString() ?? '5') ?? 5;
    dealerIndex = int.tryParse(data['dealer_index']?.toString() ?? '0') ?? 0;
   
    if (players.isEmpty && pc > 0) {
      players = List.generate(pc, (i) => Player(id: i, name: i == myMultiplayerSide ? "MOI" : "J${i+1}", hand: [], isHuman: i == myMultiplayerSide));
      for (var p in players) { sessionScores[p.id] = 0.0; }
    }

    final allCards = generateFullTarotDeck();
    final List<dynamic> cList = data['chien'] as List? ?? [];
    chien = cList.map((id) => allCards.firstWhere((card) => card.id == id)).toList();
    for (var c in chien) {
      c.assetPath = _getAssetPath(c.suit, c.value);
    }

    if (data['current_trick'] == null || (data['current_trick'] as List).isEmpty) {
      currentTrickData.clear();
      cardRotations.clear();
    }

    // --- 🕒 RECTIFICATION : SYNCHRONISATION ALIGNÉE DEPUIS TURN_ENDS_AT PLAFONNÉE À 30.0 ---
    if (data['turn_ends_at'] != null) {
      DateTime expiration = DateTime.parse(data['turn_ends_at'].toString());
      Duration reste = isMultiplayer && multiplayerManager.forcedExpiration != null ? multiplayerManager.forcedExpiration!.difference(DateTime.now()) : expiration.difference(DateTime.now());
      double secondesServeur = math.min(30.0, reste.inMilliseconds / 1000.0);

      // On ne synchronise que si l'écart est significatif (> 1.5s) pour éviter les micro-saccades
      if ((secondesServeur - remainingTime).abs() > 1.5) {
        remainingTime = secondesServeur > 0 ? secondesServeur : 0.0;
      }
    }
   
    notifyListeners();
  }

  void makeBid(BidType? b) {
    if (players.isEmpty || (players[currentPlayerIndex].isHuman && !hasClickedBid && b == null && remainingTime > 0.1)) return;
   
    // VERROU ANTI-SUBCONTRAT : Si l'enchère choisie n'est pas supérieure à l'actuelle, on force le PASSE (sauf si c'est déjà un passe)
    if (b != null && b != BidType.pass && selectedBid != null) {
      if (_bidRank(b) <= _bidRank(selectedBid)) {
        b = BidType.pass;
      }
    }

    hasClickedBid = true;
    _stopTimer();
    _applyBidLogic(b, currentPlayerIndex);

    // --- CETTE LIGNE EST LA CLÉ POUR SUPABASE ---
    if (onActionBroadcast != null) {
      onActionBroadcast!('MAKE_BID', {'bid': b.toString(), 'pid': currentPlayerIndex});
    }
    // --------------------------------------------

    _finalizeBidTurn();
    notifyListeners();
  }
  void receiveBid(String bidStr, int pid) {
    BidType b = BidType.values.firstWhere((e) => e.toString() == bidStr, orElse: () => BidType.pass);
    _applyBidLogic(b, pid);
    _finalizeBidTurn();
    notifyListeners();
  }
void _applyBidLogic(BidType? b, int pid) {
  if (b != null && b != BidType.pass) {
    if (selectedBid == null || _bidRank(b) > _bidRank(selectedBid!)) {
      selectedBid = b;
      takerId = pid;
      lastBidName = b.toString().split('.').last;
    }
  }
}
  void _finalizeBidTurn() {
    if (players.isEmpty) return;
    if (currentPlayerIndex == dealerIndex) {
      if (takerId == null) {
        dealerIndex = (dealerIndex! + 1) % (players.isEmpty ? 1 : players.length);
        if (!isMultiplayer || isHost) distributeAndStart(players.length);
      } else {
        if (players.length == 5) {
          currentState = GameState.calling;
          currentPlayerIndex = takerId!;
          _startTurnSystem();
        } else {
          _prepareDiscard();
        }
      }
    } else {
      currentPlayerIndex = (currentPlayerIndex + 1) % (players.isEmpty ? 1 : players.length);
      hasClickedBid = false;
      _startTurnSystem();
    }
  }

  bool isCardPlayable(CardModel c) => (currentState == GameState.playing && currentPlayerIndex == myMultiplayerSide) ? _isValidMove(myMultiplayerSide, c) : true;

  bool _isValidMove(int pid, CardModel c) {
    if (players.isEmpty || pid >= players.length) return false;
    var hand = players[pid].hand;
   
    if (tricksCount == 0 && currentTrickData.isEmpty && calledKingSuit != null) {
      if (c.suit == calledKingSuit) {
        if (c.value == "R") return true;
        if (!hand.every((card) => card.suit == calledKingSuit || card.suit == CardSuit.excuse)) return false;
      }
    }
    if (currentTrickData.isEmpty || c.suit == CardSuit.excuse) return true;
   
    if (leadSuit != null && hand.any((card) => card.suit == leadSuit)) {
      if (leadSuit == CardSuit.atout) {
        int mx = 0;
        for (var e in currentTrickData.where((e) => e.values.first.suit == CardSuit.atout)) {
          if (e.values.first.power > mx) mx = e.values.first.power;
        }
        if (hand.any((card) => card.suit == CardSuit.atout && card.power > mx)) {
          return c.suit == CardSuit.atout && c.power > mx;
        }
        return c.suit == CardSuit.atout;
      }
      return c.suit == leadSuit;
    }
   
    if (hand.any((card) => card.suit == CardSuit.atout)) {
      if (c.suit != CardSuit.atout) return false;
      var ot = currentTrickData.where((e) => e.values.first.suit == CardSuit.atout).toList();
      if (ot.isNotEmpty) {
        ot.sort((a, b) => b.values.first.power.compareTo(a.values.first.power));
        if (hand.any((card) => card.suit == CardSuit.atout && card.power > ot.first.values.first.power)) {
          return c.power > ot.first.values.first.power;
        }
      }
      return true;
    }
    return true;
  }

        'cardSuit': card.suit.toString(),
        'cardValue': card.value
      });

    // ----------------------------



    if (card.value == "R" && card.suit == calledKingSuit) {
      partnerId = pid;

      _updateLiveScore();

    players[pid].hand.removeWhere((c) => c.id == card.id);
    currentTrickData.add({pid: card});
   
    if (currentTrickData.length == 1) {

    } else if (leadSuit == null && card.suit != CardSuit.excuse) {

    }
   
    notifyListeners();

      // --- NOUVEAU SYSTÈME DE FLUX ---
    if (currentTrickData.length == players.length) {
      await Future.delayed(const Duration(milliseconds: 800));
      _resolveTrick();
      // --- DÉTECTION FIN DE MANCHE ---
if (players.every((p) => p.hand.isEmpty)) {
  this.currentState = GameState.scoring;
  print("🏁 Phase SCORING activée");
  notifyListeners();
  return;
  }
    } else {

}

  void _resolveTrick() async {
    if (players.isEmpty) return;
    if (leadSuit != null) {
      TarotAI.analyzeTrick(currentTrickData, leadSuit!);
    }
   
    int winner = _determineWinner();

  int _determineWinner() {
    if (currentTrickData.isEmpty) return 0;
    var trumps = currentTrickData.where((e) => e.values.first.suit == CardSuit.atout).toList();
    if (trumps.isNotEmpty) {
      trumps.sort((a, b) => b.values.first.power.compareTo(a.values.first.power));
      return trumps.first.keys.first;
    }
    var colorCards = currentTrickData.where((e) => e.values.first.suit == leadSuit).toList();
    if (colorCards.isEmpty) return currentTrickData.first.keys.first;
    colorCards.sort((a, b) => b.values.first.power.compareTo(a.values.first.power));
    return colorCards.first.keys.first;
  }

Future<void> _iaPlay() async {
  if (players.isEmpty || currentState == GameState.idle || currentState == GameState.scoring) return;

  // 1. GESTION DU TIMEOUT
  if (remainingTime <= 0.1) {
    _isProcessing = false;
  }

  // 2. SÉCURITÉ MULTIJOUEUR (LA MODIFICATION ES ICI)
  if (players[currentPlayerIndex].isHuman) {
    // Si c'est le tour d'un AUTRE joueur humain (distant), on ne fait JAMAIS rien
    if (currentPlayerIndex != myMultiplayerSide) return;
   
    // Si c'est MON tour, on ne laisse l'IA jouer que si le chrono est vraiment mort
    if (remainingTime > 0.1) return;
  }

  // 3. LOGIQUE DE JEU DE L'IA
  if (currentState == GameState.bidding) {
    makeBid(_iaSmartBidding());
  } else if (currentState == GameState.calling) {
    _iaPlayCall();
  } else if (currentState == GameState.discarding) {
    _finishDiscard();
  } else if (currentState == GameState.playing) {
    _checkPoignee(currentPlayerIndex);
    var h = players[currentPlayerIndex].hand;
    var p = h.where((c) => _isValidMove(currentPlayerIndex, c)).toList();
    playCard(currentPlayerIndex, _getBestAICard(p.isEmpty ? h : p));
  }
}





   void _startTimer() {
    _stopTimer();
    _turnTimer = Timer.periodic(const Duration(milliseconds: 100), (t) {
      if (players.isEmpty || currentState == GameState.idle || currentState == GameState.scoring) {
        _stopTimer();
        return;
      }

      // SÉCURITÉ RÉSEAU : En multijoueur, l'invité retourne directement pour éviter le conflit à 100ms
      if (isMultiplayer && !isHost) {
        return;
      }



      if (remainingTime > 0.1) {
        remainingTime -= 0.1;
      } else {
        _stopTimer();

        // 5 secondes pour l'humain, 2 secondes pour l'IA
        int secondsToWait = players[currentPlayerIndex].isHuman ? 5 : 2;

        // On ne joue que si c'est notre tour (ou solo)
        bool isMyTurn = !isMultiplayer || (currentPlayerIndex == myMultiplayerSide);

        if (isMyTurn) {
          Future.delayed(Duration(seconds: secondsToWait), () {
            if (currentState != GameState.idle && currentState != GameState.scoring) {
              _iaPlay();
              // RECTIFICATION : On utilise bien le nom original de ta fonction
              _startTurnSystem();
            }
          });
        }
        return;
      }
      notifyListeners();
    });
  }

        logic.lastBidName = ""; // On sait que celle-ci existe.
       
        // Pour le reste (Roi/Pli), on passe par le reset SQL
        // ou on attend que le MultiplayerManager écrase les données.

        _multiManager = MultiplayerManager(logic);
        String tableFinal;

        final tablesDispo = await Supabase.instance.client
            .from('table_tarot_5')
            .select('table_id')
            .eq('game_status', 'IDLE')
            .limit(1)
            .maybeSingle();

        if (tablesDispo != null && tablesDispo['table_id'] != null) {
          tableFinal = tablesDispo['table_id'] as String;
        } else {
          tableFinal = 'salle_${DateTime.now().millisecondsSinceEpoch}';
        }

        if (tableFinal.isNotEmpty) {
          _multiManager?.joinTable(tableFinal);
          RemoteLogger.send("INFO", "Entrée Jeu Online", {"table_id": tableFinal});
        }

      } else {
        logic.initGame(widget.playerCount, multi: false);
      }
    });
  }

  @override
  void dispose() {
    _multiManager?.dispose();
    super.dispose();
  }

  @override

    final logic = Provider.of<TarotGameLogic>(context);

    return Scaffold(

              icon: const Icon(Icons.arrow_back_ios, color: Color(0xFFD4AF37), size: 30),
              onPressed: () async {
                if (!context.mounted) return;
                final logicProvider = Provider.of<TarotGameLogic>(context, listen: false);
               
                if (logicProvider.version != 0) {
                  try {
                    logicProvider.lastBidName = "";
                    logicProvider.fullResetForExit();
                  } catch(_) {}
                }

                _multiManager?.dispose();
                if (context.mounted) Navigator.pop(context);
              },
            ),
          ),
        ],
      ),
    );
  }

// --- MOTEUR DE RÈGLES (Vérification de légalité) ---
class RuleEngine {
  static bool isMoveLegal(CardModel card, List<CardModel> hand, List<CardModel> trick) {
    if (card.suit == CardSuit.excuse) return true;
    if (trick.isEmpty) return true;
   
    CardSuit requestedSuit = (trick.first.suit == CardSuit.excuse && trick.length > 1)
        ? trick[1].suit
        : trick.first.suit;
       
    bool hasSuit = hand.any((c) => c.suit == requestedSuit);
    if (hasSuit) {
      if (card.suit != requestedSuit) return false;
      return true;
    }
   
    bool hasAtout = hand.any((c) => c.suit == CardSuit.atout);
    if (requestedSuit != CardSuit.atout && hasAtout) {
      return card.suit == CardSuit.atout;
    }
    return true;
  }
}
